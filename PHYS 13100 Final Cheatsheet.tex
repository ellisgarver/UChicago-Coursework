% KINEMATICS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Kinematics}
\hrule
\begin{flalign*}
& v = v_0 + a t &\\
& x = x_0 + v_0 t + \frac12 a t^2 &\\
& v^2 = v_0^2 + 2 a (x-x_0) &\\
& x - x_0 = \frac{v+v_0}{2} t &\\
& a_c = \frac{v^2}{r} = \omega^2 r &\\
& v = \frac{2\pi r}{T}, \quad a_c = \frac{4 \pi^2 r}{T^2} &\\
& x(t) = x_0 + v_{0x} t, \quad y(t) = y_0 + v_{0y} t - \frac12 g t^2 &\\
& v_x = v_0 \cos\theta, \quad v_y = v_0 \sin\theta - g t &\\
& R = \frac{v_0^2 \sin 2\theta}{g}, \quad T = \frac{2 v_0 \sin\theta}{g} &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% FORCES
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Forces}
\hrule
\begin{flalign*}
& \sum \vec F = m \vec a &\\
& F_g = mg, \quad F_s = -kx &\\
& f_s \le \mu_s N, \quad f_k = \mu_k N &\\
& N = mg \cos\theta, \quad F_{\parallel} = mg \sin\theta &\\
& v_{\max} = \sqrt{\frac{rg(\sin\theta+\mu \cos\theta)}{\cos\theta-\mu \sin\theta}} &\\
& v_{\min} = \sqrt{\frac{rg(\sin\theta-\mu \cos\theta)}{\cos\theta+\mu \sin\theta}} &\\
& F_{\text{centripetal}} = m \frac{v^2}{r} = m r \omega^2 &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% WORK / ENERGY / POWER
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Work, Energy, Power}
\hrule
\begin{flalign*}
& W = \vec F \cdot \vec d = F d \cos\theta &\\
& K = \frac12 m v^2, \quad U_g = m g h, \quad U_s = \frac12 k x^2 &\\
& E = K + U, \quad K_i + U_i = K_f + U_f &\\
& W_{\text{net}} = \Delta K, \quad P = \frac{dW}{dt} = \vec F \cdot \vec v &\\
& K_{\text{orbit}} = \frac{G M m}{2 r}, \quad E_{\text{orbit}} = -\frac{G M m}{2 r} &\\
& v = \sqrt{\frac{2(E-U)}{m}} &\\
& v = \sqrt{2 g h} \quad (\text{from } K = U) &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% MOMENTUM & IMPULSE
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Momentum \& Impulse}
\hrule
\begin{flalign*}
& \vec p = m \vec v &\\ & \vec J = \Delta \vec p = \int \vec F dt &\\
& \sum \vec p_i = \sum \vec p_f &\\
& \vec F_{\text{avg}} \Delta t = \Delta \vec p &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% COLLISIONS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Collisions}
\hrule
\begin{flalign*}
& \text{Elastic (1D): } v_1' = \frac{(m_1-m_2)v_1 + 2 m_2 v_2}{m_1+m_2}, \quad v_2' = \frac{(m_2-m_1)v_2 + 2 m_1 v_1}{m_1+m_2} &\\
& \text{Inelastic: momentum conserved, KE not conserved} &\\
& \text{Perfectly inelastic: } v_f = \frac{m_1 v_1 + m_2 v_2}{m_1 + m_2} &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% ROTATION / TORQUE / ACCELERATIONS
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rotation / Torque / Acceleration}
\hrule
\begin{flalign*}
& \omega = \omega_0 + \alpha t, \quad \theta = \theta_0 + \omega_0 t + \frac12 \alpha t^2 &\\
& \omega^2 = \omega_0^2 + 2 \alpha (\theta - \theta_0) &\\
& \tau = r F \sin\theta = I \alpha &\\
& K_{\text{rot}} = \frac12 I \omega^2, \quad L = I \omega, \quad \tau = \frac{dL}{dt} &\\
& I_{\text{disk}} = \frac12 M R^2, \quad I_{\text{shell}} = M R^2, \quad I_{\text{sphere}} = \frac{2}{5} M R^2 &\\
& I_{\text{rod, center}} = \frac{1}{12} M L^2, \quad I_{\text{rod, end}} = \frac{1}{3} M L^2 &\\
& a_t = r \alpha \quad (\text{tangential}), \quad a_r = \frac{v^2}{r} = r \omega^2 \quad (\text{radial/centripetal}) &\\
& a_{\text{total}} = \sqrt{a_t^2 + a_r^2}, \quad v_{\text{cm}} = R \omega \quad (\text{rolling without slipping}) &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% OSCILLATIONS / SHM
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Oscillations / SHM}
\hrule
\begin{flalign*}
& F = -k x, \quad T = 2\pi \sqrt{\frac{m}{k}}, \quad \omega = \sqrt{\frac{k}{m}} &\\
& x(t) = A \cos(\omega t + \phi), \quad v(t) = -A \omega \sin(\omega t + \phi) &\\
& a(t) = -A \omega^2 \cos(\omega t + \phi), \quad E = \frac12 k A^2 = K + U &\\
& T_{\text{pendulum}} = 2 \pi \sqrt{\frac{L}{g}}, \quad \omega_{\text{pendulum}} = \sqrt{\frac{g}{L}} &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% GRAVITATION MINI-BLOCK
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Gravitation}
\hrule
\begin{flalign*}
& F_g = \frac{G M m}{r^2}, \quad U = - \frac{G M m}{r} &\\
& v_{\text{orbit}} = \sqrt{\frac{G M}{r}}, \quad T = \frac{2\pi r}{v} = 2\pi \sqrt{\frac{r^3}{G M}} &\\
& K_{\text{orbit}} = \frac{G M m}{2 r}, \quad E_{\text{total}} = -\frac{G M m}{2 r} &\\
& v_{\text{escape}} = \sqrt{\frac{2 G M}{r}}, \quad T_{\text{orbit}} = 2 \pi \sqrt{\frac{r^3}{GM}} &\\
& a_r = \frac{v^2}{r} = \frac{GM}{r^2} &
\end{flalign*}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% SPECIAL RELATIVITY
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Relativity}
\hrule
\begin{flalign*}
& \gamma = \frac{1}{\sqrt{1 - v^2/c^2}} &\\
& \Delta t = \gamma \Delta t_0, \quad L = \frac{L_0}{\gamma} &\\
& p = \gamma m v, \quad E = \gamma m c^2, \quad K = (\gamma - 1) m c^2 &\\
& E^2 = (pc)^2 + (m c^2)^2 &
\end{flalign*}
