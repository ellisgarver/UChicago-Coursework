\noindent PHYS 131000 formulas, concepts, and notes.
\\ Basic formulas:
\begin{enumerate}
    \item K1: $v_x=v_{x_0}+a_xt\Rightarrow\omega=\omega_0+\alpha t$
    \item K2: $x=x_0+v_{x_0}t+\frac{1}{2}a_xt^2\Rightarrow\theta=\theta_0+\omega_0t+\frac{1}{2}\alpha t$
    \item K3: $v_x^2=v_{x_0}^2+2a_x(x-x_0)$
    \item Force: $F=ma\Rightarrow\vec{F}=\frac{dp}{dt}\Rightarrow\vec{a}=\frac{\vec{F_{net}}}{m}$
    \item Momentum: $\vec{p}=m\vec{v}\Rightarrow\vec{L}=\vec{r}\times\vec{p}=I\omega$
    \item Kinetic Energy: $K=\frac{1}{2}mv^2\Rightarrow K=\frac{1}{2}I\omega^2$
    \item Potential Energy: $U=mgh$
    \item Work: $\Delta E=W=\int\vec{F}\cdot d\vec{r}$
    \item Power: $\vec{J}=\int\vec{F}dt=\Delta\vec{p}$
    \item Impulse: $P=\frac{dE}{dt}=\vec{F}\cdot\vec{v}$
    \item Rotational: $a_c=\frac{v^2}{r}=\omega^2r,\; v=r\omega,\; \vec{\alpha}=\frac{\vec{\tau}_{net}}{I},\; I=\int r^2dm=\sum mr^2,\; T=\frac{2\pi}{\omega}=\frac{1}{f}$
    \item Spring: $\vec{F}_s=-k\Delta\vec{x},\; U_s=\frac{1}{2}k(\Delta x)^2,\; T_s=2\pi\sqrt{\frac{m}{k}}$
    \item Pendulum: $T_p=2\pi\sqrt{\frac{l}{g}}$
    \item Astro: $|\vec{F}_G|=\frac{Gm_1m_2}{r^2},\; U_G=-\frac{Gm_1m_2}{r}$

\end{enumerate}
